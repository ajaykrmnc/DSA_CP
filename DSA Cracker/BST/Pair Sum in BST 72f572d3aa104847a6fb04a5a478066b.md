# Pair Sum in BST

```cpp
//Function to check if any pair exists in BST whose sum is equal to given value.
bool helper(Node *root, int x, unordered_set<int> &s){
    if(root==NULL) return false;
    if(helper(root->left, x, s)) return true;
    if(s.find(x-root->data)!=s.end()) return true;
    else s.insert(root->data);
    return helper(root->right, x, s);
}

bool findPair(Node* root, int x) 
{
    // Your code here
    if(root==NULL) return false;
    unordered_set<int> s;
    return helper(root, x, s);
    
}
```
